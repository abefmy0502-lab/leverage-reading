```bash
ln -s /path/to/本体/node_modules node_modules       # worktree のときだけ（終わったら rm node_modules）
VITE_DEMO=true npx vite --port 5198                  # 別のターミナル（npm run demo でも可・そのときは URL を 5173 に）
UI_SHOTS_URL=http://localhost:5198 node scripts/appstore-shots.mjs   # 撮る → 重ねる（30 枚）
```
